# 桌面流程图编辑器 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use `superpowers:subagent-driven-development` (recommended) or `superpowers:executing-plans` to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 交付一个支持 Windows/macOS 的 Visio 风格流程图桌面编辑器首期版本。

**Architecture:** 文档、X6 Cell 与命令快照统一使用 pt 逻辑坐标；Vue/X6 只负责交互与渲染，Tauri/Rust 提供安全的文件、导出与本地存储能力。用户可见撤销栈由领域命令驱动，而不是 X6 属性变更历史。

**Tech Stack:** Tauri 2、Rust、Vue 3、TypeScript、Vite、Vitest、Playwright、AntV X6、Pinia、SQLite。

## Global Constraints

- 仅支持 Windows 10/11 与 macOS（Intel/Apple Silicon）。
- 图文件、X6 Cell 和导出逻辑统一以 pt 保存；CSS px 仅限渲染层。
- `devicePixelRatio` 不进入文档、命中检测和撤销数据。
- 一个文件只有一个清晰职责；不得创建聚合无关逻辑的 `utils.ts` 或 `editor-actions.ts`。
- 每个用户行为只生成一条撤销/重做记录；X6 内置 History 不作为用户历史。
- 当前阶段不实现泳道、Linux、多人协作、VSDX/BPMN 兼容和复杂全局避障。

```typescript
// src/domain/measurement.ts
export type Unit = 'mm' | 'cm' | 'in' | 'pt' | 'px'

const PT_PER_INCH = 72
const CM_PER_INCH = 2.54
const MM_PER_INCH = 25.4
const PX_PER_INCH = 96

export function unitToPt(value: number, unit: Unit): number {
  switch (unit) {
    case 'pt': return value
    case 'in': return value * PT_PER_INCH
    case 'cm': return value * (PT_PER_INCH / CM_PER_INCH)
    case 'mm': return value * (PT_PER_INCH / MM_PER_INCH)
    case 'px': return value * (PT_PER_INCH / PX_PER_INCH)
    default: throw new Error(`Unknown unit: ${unit}`)
  }
}

export function ptToUnit(pt: number, unit: Unit): number {
  switch (unit) {
    case 'pt': return pt
    case 'in': return pt / PT_PER_INCH
    case 'cm': return pt * (CM_PER_INCH / PT_PER_INCH)
    case 'mm': return pt * (MM_PER_INCH / PT_PER_INCH)
    case 'px': return pt * (PX_PER_INCH / PT_PER_INCH)
    default: throw new Error(`Unknown unit: ${unit}`)
  }
}
```

```typescript
// src/infrastructure/x6/viewport-transform.ts
export class ViewportTransform {
  private readonly PT_TO_PX_RATIO = 96 / 72
  
  documentPtToScreenPx(pt: number, zoom: number): number {
    return pt * this.PT_TO_PX_RATIO * zoom
  }
  
  screenPxToDocumentPt(px: number, zoom: number): number {
    return px / (this.PT_TO_PX_RATIO * zoom)
  }
}
```

```typescript
const ALLOWED_PROTOCOLS = ['http:', 'https:', 'mailto:']

export function isValidHyperlink(url: string): boolean {
  try {
    const parsed = new URL(url)
    return ALLOWED_PROTOCOLS.includes(parsed.protocol)
  } catch {
    return false
  }
}

export function validateHyperlink(url: string): string | null {
  if (!isValidHyperlink(url)) {
    return '仅支持 http、https、mailto 链接。'
  }
  return null
}
```

```typescript
// src/application/search/find-text.ts
import { DiagramDocument, FindTextRequest, FindMatch } from '@/domain/diagram'

export function findText(document: DiagramDocument, request: FindTextRequest): FindMatch[] {
  const matches: FindMatch[] = []
  
  for (const page of document.pages) {
    if (request.scope === 'currentPage' && page.id !== request.currentPageId) {
      continue
    }
    
    // Search nodes
    for (const node of page.nodes) {
      if (node.text?.value.includes(request.query)) {
        matches.push({
          pageId: page.id,
          cellId: node.id,
          field: 'nodeText',
          start: node.text.value.indexOf(request.query),
          end: node.text.value.indexOf(request.query) + request.query.length
        })
      }
    }
    
    // Search edges
    for (const edge of page.edges) {
      if (edge.label?.value.includes(request.query)) {
        matches.push({
          pageId: page.id,
          cellId: edge.id,
          field: 'edgeLabel',
          start: edge.label.value.indexOf(request.query),
          end: edge.label.value.indexOf(request.query) + request.query.length
        })
      }
    }
  }
  
  return matches
}
```
